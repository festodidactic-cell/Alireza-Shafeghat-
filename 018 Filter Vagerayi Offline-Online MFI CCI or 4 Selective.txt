# فیلتر 14 — DIVE-MC

## **Divergence Intelligence — MFI & CCI**

این فیلتر اختصاصی **واگرایی مثبت MFI و CCI** است.

اما یک اصلاح فنی مهم:

من نمی‌خواهم صرفاً بگوییم:

```text
MFI پایین‌تر ←→ قیمت پایین‌تر
CCI پایین‌تر ←→ قیمت پایین‌تر
```

چون این لزوماً واگرایی معتبر نیست.

فیلتر باید **دو کف ساختاری قیمت** را پیدا کند و سپس مقدار اندیکاتور را در همان نقاط مقایسه کند.

یعنی:

### واگرایی مثبت معمولی

```text
قیمت:

Low 1
   \       Low 2
    \     /
     \___/
       ↓
    Low پایین‌تر


MFI / CCI:

Low 1
   \ 
    \___
        \__
          ↑
      Low بالاتر
```

یعنی:

**قیمت کف پایین‌تر ساخته ولی MFI/CCI کف بالاتری ساخته است.**

---

### واگرایی مثبت مخفی

اینجا برعکس:

```text
قیمت:

Low 1       Low 2
  \          /
   \        /
    \______/
       ↑
   کف بالاتر


MFI / CCI:

High
  \
   \____
        \
         ↓
      کف پایین‌تر
```

یعنی:

**قیمت کف بالاتری ساخته، ولی اندیکاتور کف پایین‌تری ساخته است.**

این نوع واگرایی برای **ادامه روند صعودی** می‌تواند مهم باشد.

```javascript
{
    /*
    ============================================================
    DIVE-MC
    Divergence Intelligence — MFI & CCI

    هدف:
    کشف واگرایی مثبت معمولی و مثبت مخفی
    در MFI و CCI

    کاربرد:
    WATCHLIST ONLY

    این فیلتر دستور خرید یا فروش نیست.
    ============================================================
    */

    if(typeof [ih]=="undefined")
        return false;

    if([ih].length<35)
        return false;


    // ------------------------------------------------------------
    // ساخت آرایه داده‌ها
    // ------------------------------------------------------------

    var len=Math.min([ih].length,60);

    var P=[];
    var H=[];
    var L=[];
    var V=[];
    var TP=[];

    for(var i=0;i<len;i++)
    {
        if(typeof [ih][i]=="undefined")
            return false;

        P[i]=[ih][i].PClosing;
        H[i]=[ih][i].PriceMax;
        L[i]=[ih][i].PriceMin;
        V[i]=[ih][i].QTotTran5J;

        if(H[i]<=0 || L[i]<=0 || P[i]<=0)
            return false;

        TP[i]=(H[i]+L[i]+P[i])/3;
    }


    // ------------------------------------------------------------
    // EMA
    // ------------------------------------------------------------

    var EMA=function(arr,period)
    {
        var out=[];
        var alpha=2/(period+1);

        out[0]=arr[0];

        for(var i=1;i<arr.length;i++)
            out[i]=
                alpha*arr[i]+
                (1-alpha)*out[i-1];

        return out;
    };


    // ------------------------------------------------------------
    // MFI
    // ------------------------------------------------------------

    var MFI=[];

    for(var i=0;i<len;i++)
    {
        var pos=0;
        var neg=0;

        var start=Math.max(0,i-13);

        for(var j=start;j<=i;j++)
        {
            if(j==len-1)
                continue;

            /*
               چون آرایه از امروز به گذشته است،
               j+1 روز قدیمی‌تر است.
            */

            if(TP[j]>TP[j+1])
                pos+=TP[j]*V[j];

            if(TP[j]<TP[j+1])
                neg+=TP[j]*V[j];
        }

        if(neg==0)
            MFI[i]=100;
        else
        {
            var mr=pos/neg;
            MFI[i]=100-(100/(1+mr));
        }
    }


    // ------------------------------------------------------------
    // CCI
    // ------------------------------------------------------------

    var CCI=[];

    for(var i=0;i<len;i++)
    {
        var start=Math.max(0,i-19);

        var sum=0;
        var count=0;

        for(var j=start;j<=i;j++)
        {
            sum+=TP[j];
            count++;
        }

        var sma=sum/count;

        var dev=0;

        for(var j=start;j<=i;j++)
            dev+=Math.abs(TP[j]-sma);

        dev=dev/count;

        if(dev==0)
            CCI[i]=0;
        else
            CCI[i]=(TP[i]-sma)/(0.015*dev);
    }


    // ------------------------------------------------------------
    // Pivot Low
    // ------------------------------------------------------------

    var pivotLow=function(i)
    {
        if(i<3 || i>len-4)
            return false;

        return(
            L[i]<=L[i-1] &&
            L[i]<=L[i-2] &&
            L[i]<=L[i+1] &&
            L[i]<=L[i+2]
        );
    };


    // ------------------------------------------------------------
    // آخرین دو Pivot Low معتبر
    // ------------------------------------------------------------

    var recent=-1;
    var old=-1;

    for(var i=3;i<len-3;i++)
    {
        if(pivotLow(i))
        {
            if(recent==-1)
                recent=i;
            else
            {
                old=i;
                break;
            }
        }
    }

    if(recent==-1 || old==-1)
        return false;


    // ------------------------------------------------------------
    // فاصله دو Pivot
    // ------------------------------------------------------------

    var separation=old-recent;

    if(separation<5)
        return false;


    // ------------------------------------------------------------
    // واگرایی MFI
    // ------------------------------------------------------------

    var mfiRegular=false;
    var mfiHidden=false;

    if(
        L[recent]<L[old] &&
        MFI[recent]>MFI[old]
    )
        mfiRegular=true;


    if(
        L[recent]>L[old] &&
        MFI[recent]<MFI[old]
    )
        mfiHidden=true;


    // ------------------------------------------------------------
    // واگرایی CCI
    // ------------------------------------------------------------

    var cciRegular=false;
    var cciHidden=false;


    if(
        L[recent]<L[old] &&
        CCI[recent]>CCI[old]
    )
        cciRegular=true;


    if(
        L[recent]>L[old] &&
        CCI[recent]<CCI[old]
    )
        cciHidden=true;


    // ------------------------------------------------------------
    // قدرت واگرایی
    // ------------------------------------------------------------

    var mfiStrength=0;
    var cciStrength=0;


    if(mfiRegular)
        mfiStrength=2;

    if(mfiHidden)
        mfiStrength=1;

    if(cciRegular)
        cciStrength=2;

    if(cciHidden)
        cciStrength=1;


    // ------------------------------------------------------------
    // تأیید برگشت فعلی
    // ------------------------------------------------------------

    var recovery=false;

    if(
        P[0]>P[1] ||
        P[0]>P[2] ||
        P[0]>P[3]
    )
        recovery=true;


    // ------------------------------------------------------------
    // جلوگیری از واگرایی‌های بسیار ضعیف
    // ------------------------------------------------------------

    var priceDistance=
        Math.abs(L[recent]-L[old])/
        L[old];

    /*
       اختلاف بسیار جزئی را حذف نمی‌کنیم،
       چون هدف Watchlist است.
       فقط اگر عملاً دو کف یکسان باشند،
       سیگنال را بی‌ارزش فرض می‌کنیم.
    */

    if(priceDistance<0.005)
    {
        /*
           اگر اندیکاتور اختلاف قابل توجهی دارد،
           هنوز اجازه خروجی می‌دهیم.
        */

        if(
            Math.abs(MFI[recent]-MFI[old])<4 &&
            Math.abs(CCI[recent]-CCI[old])<20
        )
            return false;
    }


    // ------------------------------------------------------------
    // نتیجه
    // ------------------------------------------------------------

    /*
       حداقل یکی از دو اندیکاتور باید واگرایی معتبر بدهد.

       MFI + CCI همزمان:
       سیگنال قوی‌تر.

       فقط یکی:
       همچنان برای Watchlist معتبر.
    */

    if(
        (mfiRegular || mfiHidden ||
         cciRegular || cciHidden)
    )
        return true;


    return false;

}()
```

# فیلتر 15 — DIVE-X

## **Multi-Divergence Radar**

این یکی همان فیلتر دوم موردنظر توست.

اما اینجا من یک تصمیم مهم گرفته‌ام:

### قرار نیست MFI + CCI + RSI + MACD + Stoch + Volume همگی همزمان واگرایی داشته باشند.

این کار اشتباه است.

اگر چنین شرطی بگذاریم:

```text
MFI divergence
AND
CCI divergence
AND
RSI divergence
AND
MACD divergence
AND
Stoch divergence
AND
Volume divergence
```

ممکن است فیلتر فوق‌العاده‌ای به نظر برسد، ولی در عمل احتمالاً **بیش از حد نادر** می‌شود.

من به‌جای آن از مفهوم **Confluence** استفاده می‌کنم:

> چند اندیکاتور مستقل، یک پیام مشابه بدهند؛ اما لازم نیست همه آنها همزمان تأیید کنند.

در نسخه‌ای که می‌نویسم، **حداقل یک واگرایی معتبر کافی است**؛ اما هرچه تعداد اندیکاتورهای همسو بیشتر شود، اهمیت سیگنال بیشتر می‌شود.

1. **MFI**
2. **CCI**

بنابراین اگر مثلاً ۴ مورد از ۶ مورد در یک ناحیه واگرایی مثبت بدهند، برای Watchlist بسیار جالب‌تر از یک واگرایی منفرد است.

# کد DIVE-X

```javascript

true==function()
{
    /*
    ================================================================
    DIVE-X
    MULTI-DIVERGENCE RADAR

    هدف:
    کشف همگرایی واگرایی‌های مثبت در چند اندیکاتور.

    کاربرد:
    WATCHLIST

    این فیلتر دستور خرید نیست.

    اندیکاتورها:
    MFI
    CCI
    RSI
    STOCHASTIC
    MACD HISTOGRAM
    VOLUME
    ================================================================
    */


    if(typeof [ih]=="undefined")
        return false;

    if([ih].length<40)
        return false;


    // ================================================================
    // DATA
    // ================================================================

    var len=Math.min([ih].length,60);

    var P=[];
    var H=[];
    var L=[];
    var V=[];
    var TP=[];

    for(var i=0;i<len;i++)
    {
        if(typeof [ih][i]=="undefined")
            return false;

        P[i]=[ih][i].PClosing;
        H[i]=[ih][i].PriceMax;
        L[i]=[ih][i].PriceMin;
        V[i]=[ih][i].QTotTran5J;

        if(P[i]<=0 || H[i]<=0 || L[i]<=0)
            return false;

        TP[i]=(H[i]+L[i]+P[i])/3;
    }


    // ================================================================
    // EMA
    // ================================================================

    var EMA=function(arr,period)
    {
        var out=[];
        var alpha=2/(period+1);

        out[0]=arr[0];

        for(var i=1;i<arr.length;i++)
            out[i]=
                alpha*arr[i]+
                (1-alpha)*out[i-1];

        return out;
    };


    // ================================================================
    // RSI
    // ================================================================

    var RSI=[];

    for(var i=0;i<len;i++)
    {
        var gain=0;
        var loss=0;

        var start=Math.min(len-1,i+13);

        for(var j=i;j<start;j++)
        {
            var ch=P[j]-P[j+1];

            if(ch>0)
                gain+=ch;
            else
                loss-=ch;
        }

        if(loss==0)
            RSI[i]=100;
        else
        {
            var rs=gain/loss;
            RSI[i]=100-(100/(1+rs));
        }
    }


    // ================================================================
    // MFI
    // ================================================================

    var MFI=[];

    for(var i=0;i<len;i++)
    {
        var pos=0;
        var neg=0;

        var start=Math.min(len-1,i+13);

        for(var j=i;j<start;j++)
        {
            var money=TP[j]*V[j];

            if(TP[j]>TP[j+1])
                pos+=money;

            if(TP[j]<TP[j+1])
                neg+=money;
        }

        if(neg==0)
            MFI[i]=100;
        else
        {
            var ratio=pos/neg;
            MFI[i]=100-(100/(1+ratio));
        }
    }


    // ================================================================
    // CCI
    // ================================================================

    var CCI=[];

    for(var i=0;i<len;i++)
    {
        var start=Math.min(len-1,i+19);

        var sum=0;
        var count=0;

        for(var j=i;j<=start;j++)
        {
            sum+=TP[j];
            count++;
        }

        var mean=sum/count;

        var dev=0;

        for(var j=i;j<=start;j++)
            dev+=Math.abs(TP[j]-mean);

        dev=dev/count;

        if(dev==0)
            CCI[i]=0;
        else
            CCI[i]=(TP[i]-mean)/(0.015*dev);
    }


    // ================================================================
    // STOCHASTIC %K
    // ================================================================

    var STOCH=[];

    for(var i=0;i<len;i++)
    {
        var start=Math.min(len-1,i+13);

        var highest=H[i];
        var lowest=L[i];

        for(var j=i;j<=start;j++)
        {
            if(H[j]>highest)
                highest=H[j];

            if(L[j]<lowest)
                lowest=L[j];
        }

        if(highest==lowest)
            STOCH[i]=50;
        else
            STOCH[i]=
                100*
                (P[i]-lowest)/
                (highest-lowest);
    }


    // ================================================================
    // MACD HISTOGRAM
    // ================================================================

    var ema12=EMA(P,12);
    var ema26=EMA(P,26);

    var macd=[];

    for(var i=0;i<len;i++)
        macd[i]=ema12[i]-ema26[i];

    var signal=EMA(macd,9);

    var MACDH=[];

    for(var i=0;i<len;i++)
        MACDH[i]=macd[i]-signal[i];


    // ================================================================
    // VOLUME OSCILLATOR
    // ================================================================

    var VOL=[];

    for(var i=0;i<len;i++)
    {
        var start5=Math.min(len-1,i+4);
        var start20=Math.min(len-1,i+19);

        var s5=0;
        var c5=0;

        var s20=0;
        var c20=0;

        for(var j=i;j<=start5;j++)
        {
            s5+=V[j];
            c5++;
        }

        for(var j=i;j<=start20;j++)
        {
            s20+=V[j];
            c20++;
        }

        var av5=s5/c5;
        var av20=s20/c20;

        if(av20==0)
            VOL[i]=0;
        else
            VOL[i]=(av5/av20)*100;
    }


    // ================================================================
    // PIVOT LOW
    // ================================================================

    var pivotLow=function(i)
    {
        if(i<3 || i>len-4)
            return false;

        return(
            L[i]<=L[i-1] &&
            L[i]<=L[i-2] &&
            L[i]<=L[i+1] &&
            L[i]<=L[i+2]
        );
    };


    // ================================================================
    // TWO RECENT PIVOTS
    // ================================================================

    var recent=-1;
    var old=-1;
```

## DIVE-MC

**فقط:**

> MFI + CCI

و به دنبال:

* واگرایی مثبت معمولی MFI
* واگرایی مثبت مخفی MFI

# DIVE-X

نسخه گسترده‌تر است:

> **MFI + CCI + RSI + Stochastic + MACD + Volume**

ولی مهم‌تر اینکه **AND سخت ندارد.**

مثلاً این کاملاً قابل قبول است:

```text
MFI       ✅
CCI       ✅
RSI       ❌
Stoch     ❌
MACD      ❌
Volume    ❌
```

و سهم وارد Watchlist می‌شود.

اما این حالت:

```text
MFI       ✅
CCI       ✅
RSI       ✅
Stoch     ❌
MACD      ✅
Volume    ❌
```

از نظر من بسیار جالب‌تر است.

و این:

```text
MFI       ✅
CCI       ✅
```

یک **Confluence بسیار غیرعادی** محسوب می‌شود و باید در بررسی دستی اولویت بالایی بگیرد.

# یک نکته مهم درباره Volume

در DIVE-X، «واگرایی حجم» را به معنای یک Volume Oscillator ساده گرفته‌ام، نه اینکه ادعا کنم این دقیقاً همان Volume Divergence حرفه‌ای نرم‌افزارهای تخصصی است.

این تفکیک مهم است.

داده خام سایت برای حجم، تعداد معاملات، ارزش معاملات، قیمت‌های روزانه و سابقه معاملات در دسترس است؛ بنابراین چنین شاخصی را می‌توان از `[ih]` ساخت. 

همچنین سایت اجازه محاسبه اندیکاتورها با کدنویسی روی `[ih]` را می‌دهد؛ مستندات خود فایل نمونه کامل محاسبه RSI را ارائه کرده‌اند.
